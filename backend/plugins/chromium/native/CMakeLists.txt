cmake_minimum_required(VERSION 3.21)
project(golet_chromium LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(USE_SANDBOX ON CACHE BOOL "CEF sandbox" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} libcef_dll_wrapper)
if(APPLE)
  add_executable(golet-chromium main_mac.mm shell_mac.mm host.cc)
  set_source_files_properties(shell_mac.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  add_executable(golet-chromium-helper helper_mac.cc)
  foreach(target golet-chromium golet-chromium-helper)
    SET_EXECUTABLE_TARGET_PROPERTIES(${target})
    target_link_libraries(${target} libcef_dll_wrapper ${CEF_STANDARD_LIBS})
  endforeach()
else()
  message(FATAL_ERROR "The CEF host packaging currently supports macOS. Windows/Linux native hosts are not yet implemented.")
endif()
